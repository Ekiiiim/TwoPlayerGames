      requirePhase(state, 'ready');
      const ready = { ...state.ready, [action.player]: true };
      if (!(ready.p1 && ready.p2)) return { ...state, ready };
      if (state.readyNext === 'preview') {

      requirePhase(state, 'preview');
      return {
        ...state, phase: 'countdown',
        active: null, selection: [], revealedCells: [],
        deadline: ctx.now + ctx.durations.countdownMs,
      };
    }
    case 'COUNTDOWN_DONE': {
      requirePhase(state, 'countdown');
      return {
        ...state, phase: 'buzzing',
        target: generateTarget(state.board),
        active: null, selection: [], revealedCells: [], deadline: null,
      };
    }
    case 'REVEAL_DONE': {
      requirePhase(state, 'reveal');
      const revealIndex = (state.revealIndex + 1) % state.board.length;
      return {
        ...state, phase: 'countdown', revealIndex,
        active: null, selection: [], revealedCells: [],
        deadline: ctx.now + ctx.durations.countdownMs,
      };

**Files:** `packages/server/src/gameSession.ts`、`src/index.ts`、`test/integration.test.ts`

- [ ] **Step 1: 改集成测试(新流程)**

`test/integration.test.ts`:
- `FAST` 加 `countdownMs`:`const FAST = { previewMs: 40, answerMs: 80, revealMs: 20, resolveMs: 20, countdownMs: 20 };`
- `createJoin` 改为:加入后两端都收到首个 view(`phase==='ready'`),然后两端各 `emit('ready')`:
```ts
async function createJoin(port: number) {
  const a = connect(port);
  const b = connect(port);
  a.emit('create_room');
  const { roomCode } = await once<{ roomCode: string }>(a, 'room_created');
  const pA = once<ClientView>(a, 'view_update');
  const pB = once<ClientView>(b, 'view_update');
  b.emit('join_room', { roomCode });
  await pA; await pB; // 双方进入 ready
  return { a, b, roomCode };
}
// 双方点准备,推进过 preview/countdown 直到 buzzing
async function readyToBuzzing(a: Socket, b: Socket) {
  a.emit('ready'); b.emit('ready');
  return waitView(a, (v) => v.phase === 'buzzing');
}
```
- 第一个用例改名/改为验证 ready→…→buzzing,并断言 target 仅 buzzing 出现:
```ts
  it('join -> ready; both ready -> preview/countdown -> buzzing (target only at buzzing)', async () => {
    const { port, close } = await startServer(0, { durations: FAST });
    stop = close;
    const { a, b } = await createJoin(port);
    const r = await once<ClientView>(a, 'view_update'); // 已在 ready
    // ready 阶段 target 为 null
    const first = await waitView(a, () => true);
    expect(['ready']).toContain(first.phase === 'ready' ? 'ready' : first.phase);
    const buzzing = await readyToBuzzing(a, b);
    expect(buzzing.target).not.toBeNull();
    a.close(); b.close();
  });
```
> 注:上面 `r`/`first` 仅示意;实现时以"等到 phase==='buzzing' 且 target 非空"为核心断言,并可加"在 buzzing 之前任意 view 的 target 都为 null"的收集式断言。
- 把所有原先"createJoin 后直接 `waitView(phase==='buzzing')`"的用例,改为 `await readyToBuzzing(a,b)`。
- "play to a win" 循环:每回合开始若 `phase==='ready'` 则双方 `emit('ready')`,再 `waitView(phase==='buzzing')` 后抢答作答(因为答对后回到 ready 门)。
